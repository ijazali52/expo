// App.js - TikTok Coin Calculator (React Native)
import React, { useState } from 'react';
import { View, Text, TextInput, StyleSheet } from 'react-native';

const COIN_TO_USD = 0.013; // Example: 1 Coin = $0.013

export default function App() {
  const [coins, setCoins] = useState('');
  const usdValue = (parseFloat(coins) || 0) * COIN_TO_USD;

  return (
    <View style={styles.container}>
      <Text style={styles.title}>TikTok Coins Calculator</Text>
      <TextInput
        style={styles.input}
        placeholder="Enter number of coins"
        keyboardType="numeric"
        value={coins}
        onChangeText={setCoins}
      />
      <Text style={styles.result}>USD Value: ${usdValue.toFixed(2)}</Text>
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, justifyContent: 'center', alignItems: 'center' },
  title: { fontSize: 24, marginBottom: 20 },
  input: {
    height: 50,
    borderColor: '#ccc',
    borderWidth: 1,
    width: '80%',
    paddingHorizontal: 10,
    marginBottom: 20,
  },
  result: { fontSize: 18 },
});
